'use strict';
const fs=require('node:fs'),path=require('node:path'),cp=require('node:child_process'),assert=require('node:assert/strict');
const gh=path.resolve('../../validation-evidence/github-cli/bin/gh.exe'),repo='xariib/Lumenfall';
const required=['pre-merge-validation.yml','comet-expansion-validation.yml','forge-expansion-validation.yml','lab-expansion-validation.yml','shared-ui-validation.yml','tree-focused-qa.yml'];
const extra=['permanent-progression-validation.yml','wisp-scroll-validation.yml','migration-focused-validation.yml'];
function call(args,input){return cp.execFileSync(gh,args,{encoding:'utf8',input:input===undefined?undefined:JSON.stringify(input),maxBuffer:64*1024*1024});}
function api(route,input){return JSON.parse(call(['api',...(input?['--method','POST','--input','-']:[]),'repos/'+repo+'/'+route],input)||'null');}
function status(sha){return api('actions/runs?head_sha='+sha+'&per_page=100').workflow_runs.filter(r=>r.head_sha===sha);}
function latest(runs,events){const chosen=new Map();for(const r of runs.sort((a,b)=>b.id-a.id)){const key=path.basename(r.path);if((!events||events.includes(r.event))&&!chosen.has(key))chosen.set(key,r);}return chosen;}
const [action,arg,second]=process.argv.slice(2);
if(action==='status'){
 const runs=status(arg),rows=runs.map(r=>({id:r.id,path:r.path,event:r.event,status:r.status,conclusion:r.conclusion,attempt:r.run_attempt,url:r.html_url,head:r.head_sha}));
 fs.mkdirSync('docs/qa/support-duration-001/remote',{recursive:true});fs.writeFileSync('docs/qa/support-duration-001/remote/status-'+arg.slice(0,8)+'.json',JSON.stringify({sha:arg,observedAt:new Date().toISOString(),runs:rows},null,2)+'\n');
 console.log(JSON.stringify(rows.map(r=>({id:r.id,path:path.basename(r.path),event:r.event,status:r.status,conclusion:r.conclusion}))));
}else if(action==='dispatch'){
 assert(['fix/support-duration-001','main'].includes(arg));const head=api('git/ref/heads/'+arg).object.sha;
 assert.equal(head,second,'exact intended ref before dispatch');const files=process.argv[5]?[process.argv[5]]:required.concat(extra);
 for(const file of files){assert(required.concat(extra).includes(file));api('actions/workflows/'+file+'/dispatches',{ref:arg});console.log(file+': dispatched '+head);}
}else if(action==='build'){
 const head=api('git/ref/heads/main').object.sha;assert.equal(head,arg,'exact integrated main');const selected=latest(status(head),['workflow_dispatch']);
 for(const file of required)assert.equal(selected.get(file)?.conclusion,'success','integrated release gate '+file);
 assert(['true','false'].includes(second));api('actions/workflows/build-android.yml/dispatches',{ref:'main',inputs:{publish:second}});console.log('Android build dispatched for '+head+' publish='+second);
}else if(action==='run'){
 const r=api('actions/runs/'+arg);console.log(JSON.stringify({id:r.id,sha:r.head_sha,branch:r.head_branch,event:r.event,status:r.status,conclusion:r.conclusion,created:r.created_at,started:r.run_started_at,attempt:r.run_attempt,url:r.html_url,path:r.path,actor:r.actor?.login}));
}else if(action==='jobs'){
 const jobs=api('actions/runs/'+arg+'/jobs?filter=latest&per_page=100').jobs;fs.mkdirSync('docs/qa/support-duration-001/remote',{recursive:true});fs.writeFileSync('docs/qa/support-duration-001/remote/jobs-'+arg+'.json',JSON.stringify(jobs,null,2)+'\n');console.log(JSON.stringify(jobs.map(j=>({id:j.id,name:j.name,status:j.status,conclusion:j.conclusion,failed:j.steps.filter(s=>s.conclusion==='failure').map(s=>s.name)}))));
}else if(action==='capture'||action==='capture-failure'){
 assert(/^\d+$/.test(arg));const run=api('actions/runs/'+arg);assert.equal(run.conclusion,action==='capture'?'success':'failure');
 const jobs=api('actions/runs/'+arg+'/jobs?filter=latest&per_page=100').jobs,out='docs/qa/support-duration-001/remote';fs.mkdirSync(out,{recursive:true});
 const records=[];for(const job of jobs){if(action==='capture')assert.equal(job.conclusion,'success');const raw=call(['api','--allow-escape-sequences','repos/'+repo+'/actions/jobs/'+job.id+'/logs']);const name='job-'+job.id+'.log.gz';fs.writeFileSync(path.join(out,name),require('node:zlib').gzipSync(raw));records.push({id:job.id,name:job.name,conclusion:job.conclusion,started:job.started_at,completed:job.completed_at,log:name,rawSha256:require('node:crypto').createHash('sha256').update(raw).digest('hex'),summary:raw.split('\n').filter(s=>/Behavioral QA coverage:|Behavioral QA passed:|Behavioral QA result:|ALL.*PASS|190\/|190.*pass|TOTAL.*PASS/i.test(s)).map(s=>s.slice(0,500))});}
 const receipt={id:run.id,sha:run.head_sha,event:run.event,conclusion:run.conclusion,url:run.html_url,jobs:records};fs.writeFileSync(out+'/'+(action==='capture'?'passed':'failed')+'-'+arg+'.json',JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify(receipt));
}else if(action==='notes'){
 const r=api('releases/tags/android-1000128'),notes=fs.readFileSync('docs/qa/support-duration-001/release-notes.txt','utf8');assert.equal(r.body.trim(),notes.trim());assert.equal(r.target_commitish,'f16ffd9b4319ff64fe3ecfb28ddc07a881620ec8');assert.equal(r.draft,false);assert.equal(r.prerelease,false);const receipt={status:'PASS',utc:new Date().toISOString(),tag:r.tag_name,title:r.name,source:r.target_commitish,url:r.html_url,notesSha256:require('node:crypto').createHash('sha256').update(notes).digest('hex')};fs.writeFileSync('docs/qa/support-duration-001/android-release/release-notes-verification.json',JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify(receipt));
}else if(action==='download'){
 assert(/^\d+$/.test(arg));const target=path.resolve('validation-evidence/android-'+arg);fs.mkdirSync(target,{recursive:true});call(['run','download',arg,'--repo',repo,'--dir',target]);console.log(target);
}else throw Error('Unknown task delivery action');
